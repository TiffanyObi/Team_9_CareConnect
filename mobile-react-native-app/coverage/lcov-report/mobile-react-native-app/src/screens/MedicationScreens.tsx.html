import React from 'react'; import { Alert, ScrollView, StyleSheet } from 'react-native'; import { BottomTabScreenProps } from '@react-navigation/bottom-tabs'; import { NativeStackScreenProps } from '@react-navigation/native-stack'; import { AppText, Button, Card, Screen, StatusMessage } from '../components/UI'; import { useApp } from '../context/AppContext'; import { medications } from '../utils/data'; import { formatDateTime, isSameLocalDay } from '../utils/theme'; import { RootParams, TabParams } from '../navigation/RootNavigator';
export function MedicationsScreen({ navigation }: BottomTabScreenProps<TabParams, 'Medications'>): React.JSX.Element { const { medicationLogs } = useApp(); return <Screen><ScrollView contentContainerStyle={s.content}><AppText heading>Medications</AppText><AppText>Your current medication schedule</AppText>{medications.map(m => <Card key={m.id} label={`${m.name}, ${m.dosage}, ${m.schedule}`}><AppText style={s.title}>{m.name}</AppText><AppText>{m.dosage} • {m.schedule}</AppText><Button label="View details" accessibilityLabel={`View details for ${m.name}`} secondary onPress={() => navigation.getParent()?.navigate('MedicationDetail', { medication: m })} /></Card>)}<AppText accessibilityRole="header" style={s.section}>Medication logs</AppText>{medicationLogs.length === 0 ? <AppText>No doses logged yet.</AppText> : medicationLogs.map(log => <Card key={log.id}><AppText>{log.medicationName} taken</AppText><AppText>{formatDateTime(log.takenAt)}</AppText></Card>)}</ScrollView></Screen>; }
export function MedicationDetailScreen({ route }: NativeStackScreenProps<RootParams, 'MedicationDetail'>): React.JSX.Element { const { medication } = route.params; const { medicationLogs, markMedicationTaken } = useApp(); const [saving, setSaving] = React.useState(false); const latestLog = medicationLogs.find(log => log.medicationId === medication.id); const loggedToday = latestLog ? isSameLocalDay(latestLog.takenAt, new Date()) : false; const markTaken = async () => { setSaving(true); try { await markMedicationTaken(medication); } catch { Alert.alert('Medication not logged', 'Please try again.'); } finally { setSaving(false); } }; return <Screen><ScrollView contentContainerStyle={s.content}><AppText heading>{medication.name}</AppText><AppText>{medication.dosage} • {medication.schedule}</AppText><Card tone="warning"><AppText style={[s.title, s.darkText]}>Today’s dose</AppText><StatusMessage style={s.darkText} message={loggedToday ? 'Logged today' : 'Not logged today'} /><AppText style={s.darkText}>{latestLog ? `Last taken ${formatDateTime(latestLog.takenAt)}` : 'No previous doses logged'}</AppText></Card><Card><AppText style={s.title}>Instructions</AppText><AppText>{medication.instructions}</AppText></Card><Button label={saving ? 'Saving…' : 'Mark as taken'} disabled={saving} onPress={() => { void markTaken(); }} /><Card tone="info"><AppText style={[s.title, s.darkText]}>Sharing preview</AppText><AppText style={s.darkText}>Caregiver sees: taken/missed status only.</AppText></Card></ScrollView></Screen>; }
const s = StyleSheet.create({ content: { padding: 20, paddingBottom: 32, gap: 12, maxWidth: 760, width: '100%', alignSelf: 'center' }, title: { fontWeight: '700', fontSize: 18 }, section: { fontWeight: '800', fontSize: 20, marginTop: 8 }, darkText: { color: '#10233F' } });
