import React, { useRef, useState } from 'react'; import { Pressable, ScrollView, StyleSheet, View } from 'react-native'; import { BottomTabScreenProps } from '@react-navigation/bottom-tabs'; import { NativeStackScreenProps } from '@react-navigation/native-stack'; import { AppText, Button, Card, FormField, Screen } from '../components/UI'; import { AccessibleModal } from '../components/AccessibleModal'; import { messages } from '../utils/data'; import { RootParams, TabParams } from '../navigation/RootNavigator';
export function MessagesScreen({ navigation }: BottomTabScreenProps<TabParams, 'Messages'>): React.JSX.Element { const [compose, setCompose] = useState(false); const composeTrigger = useRef<React.ElementRef<typeof Pressable>>(null); return <Screen><View style={{ flex: 1 }} accessibilityElementsHidden={compose} importantForAccessibility={compose ? 'no-hide-descendants' : 'auto'}><ScrollView contentContainerStyle={s.content}><AppText heading>Care team messages</AppText><AppText secondary>Private conversations with clear sharing</AppText>{messages.map(m => <Card key={m.id}><AppText style={s.title}>{m.subject}</AppText><AppText>{m.sender}</AppText><Button label="Open message" accessibilityLabel={`Open message: ${m.subject}, from ${m.sender}`} secondary onPress={() => navigation.getParent()?.navigate('MessageDetail', { message: m })} /></Card>)}<Button label="New message" buttonRef={composeTrigger} accessibilityHint="Opens a draft form; sending is not yet available" onPress={() => setCompose(true)} /><Card tone="success"><AppText style={[s.title, s.darkText]}>Privacy note</AppText><AppText style={s.darkText}>Messages show only the sender’s name on the lock screen.</AppText></Card></ScrollView></View><AccessibleModal visible={compose} title="New message" returnFocusRef={composeTrigger} onClose={() => setCompose(false)}><FormField label="Message caregiver" multiline /><Button label="Send" disabled /></AccessibleModal></Screen>; }
export function MessageDetailScreen({ route }: NativeStackScreenProps<RootParams, 'MessageDetail'>): React.JSX.Element { const { message } = route.params; return <Screen><ScrollView contentContainerStyle={s.content}><AppText heading>{message.subject}</AppText><AppText style={{ fontWeight: '700' }}>From {message.sender}</AppText><AppText style={{ fontSize: 18, marginTop: 12 }}>{message.body}</AppText></ScrollView></Screen>; }
const s = StyleSheet.create({ content: { padding: 20, paddingBottom: 32, gap: 12, maxWidth: 760, width: '100%', alignSelf: 'center' }, title: { fontSize: 18, fontWeight: '700' }, darkText: { color: '#10233F' } });
