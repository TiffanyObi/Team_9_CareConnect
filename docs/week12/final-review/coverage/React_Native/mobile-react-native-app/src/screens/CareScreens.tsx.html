import React, { useRef, useState } from 'react';
import { Alert, Pressable, ScrollView, StyleSheet, View } from 'react-native';
import { BottomTabScreenProps } from '@react-navigation/bottom-tabs';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { AppText, Button, Card, FormField, Screen } from '../components/UI';
import { AccessibleModal } from '../components/AccessibleModal';
import { appointments } from '../utils/data'; import { RootParams, TabParams } from '../navigation/RootNavigator'; import { useApp } from '../context/AppContext'; import { formatTime } from '../utils/theme';
export function CareScreen({ navigation }: BottomTabScreenProps<TabParams, 'Care'>): React.JSX.Element {
 const [showAdd, setShowAdd] = useState(false); const addTrigger = useRef<React.ElementRef<typeof Pressable>>(null); return <Screen><View style={{ flex: 1 }} accessibilityElementsHidden={showAdd} importantForAccessibility={showAdd ? 'no-hide-descendants' : 'auto'}><ScrollView contentContainerStyle={s.content}><AppText heading>Your appointments</AppText><AppText>Upcoming visits and questions to ask</AppText>{appointments.map(a => <Card key={a.id}><AppText style={s.title}>{a.title}</AppText><AppText>{a.dateAndTime}{'\n'}{a.location}</AppText><Button label="View appointment" accessibilityLabel={`View appointment: ${a.title}`} secondary onPress={() => navigation.getParent()?.navigate('AppointmentDetail', { appointment: a })} /></Card>)}<Button label="Add appointment" buttonRef={addTrigger} accessibilityHint="Opens a draft form; saving is not yet available" onPress={() => setShowAdd(true)} /><Button label="Health log" secondary onPress={() => navigation.getParent()?.navigate('HealthLog')} /><Button label="Emergency assistance" danger onPress={() => navigation.getParent()?.navigate('Emergency')} /></ScrollView></View><AccessibleModal returnFocusRef={addTrigger} visible={showAdd} title="Add appointment" onClose={() => setShowAdd(false)}><FormField label="Appointment name" /><FormField label="Date and time" /><Button label="Save appointment" disabled /></AccessibleModal></Screen>;
}
export function AppointmentDetailScreen({ route, navigation }: NativeStackScreenProps<RootParams, 'AppointmentDetail'>): React.JSX.Element { const { appointment } = route.params; return <Screen><ScrollView contentContainerStyle={s.content}><AppText heading>{appointment.title}</AppText><AppText secondary>{appointment.dateAndTime} • {appointment.location}</AppText><Card><AppText style={s.title}>Before you go</AppText><AppText>Bring insurance card • Arrive 10 minutes early</AppText></Card><Card tone="info"><AppText style={[s.title, s.darkText]}>Questions for the therapist</AppText><AppText style={s.darkText}>{'• Is my knee pain improving?\n• Which home exercises are safest?'}</AppText></Card><Button label="Check in" onPress={() => navigation.navigate('HealthLog')} /><Button label="Message caregiver" secondary onPress={() => navigation.navigate('Tabs', { screen: 'Messages' } as never)} /></ScrollView></Screen>; }
export function HealthLogScreen(): React.JSX.Element { const { logs, addLog } = useApp(); const [symptom, setSymptom] = useState(''); const [notes, setNotes] = useState(''); const [saving, setSaving] = useState(false); const save = async () => { if (!symptom.trim()) { Alert.alert('Choose a symptom to continue.'); return; } setSaving(true); try { await addLog(symptom, notes.trim()); setSymptom(''); setNotes(''); Alert.alert('Health log saved.'); } catch { Alert.alert('Health log not saved', 'Please try again.'); } finally { setSaving(false); } }; return <Screen><ScrollView contentContainerStyle={s.content}><AppText heading>How are you feeling?</AppText><AppText>Record symptoms without animation or pressure</AppText><FormField label="Symptom" value={symptom} onChangeText={setSymptom} placeholder="Knee pain, fatigue, headache…" /><FormField label="Private notes (optional)" value={notes} onChangeText={setNotes} multiline placeholder="Add details about how you feel" /><Button label={saving ? 'Saving…' : 'Save today’s log'} disabled={saving} onPress={() => { void save(); }} /><AppText accessibilityRole="header" style={s.section}>Recent health logs</AppText>{logs.length === 0 ? <AppText>No symptoms logged yet.</AppText> : logs.map(log => <Card key={log.id}><AppText style={s.title}>{log.symptom}</AppText><AppText>{log.notes || 'No note added'}{'\n'}{formatTime(log.recordedAt)}</AppText></Card>)}</ScrollView></Screen>; }
